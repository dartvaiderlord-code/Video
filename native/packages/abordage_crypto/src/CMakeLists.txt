cmake_minimum_required(VERSION 3.15)
project(abordage_crypto LANGUAGES C)
add_library(abordage_crypto SHARED abordage_crypto.c monocypher.c)
set_target_properties(abordage_crypto PROPERTIES C_VISIBILITY_PRESET hidden)
if(MSVC)
  target_compile_options(abordage_crypto PRIVATE /O2)
else()
  target_compile_options(abordage_crypto PRIVATE -O3 -fvisibility=hidden)
endif()
